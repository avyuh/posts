    <script>
      (function () {
        const shell = document.querySelector(".table-shell");
        const scrollArea = document.querySelector(".table-scroll");
        const tbody = document.getElementById("apartment-body");
        const headers = Array.from(document.querySelectorAll("thead th"));
        const rows = Array.from(tbody.querySelectorAll("tr"));
        const state = { key: "price", direction: "asc" };

        function updateScrollHints() {
          const maxScroll = scrollArea.scrollWidth - scrollArea.clientWidth;
          shell.dataset.left = scrollArea.scrollLeft > 6 ? "true" : "false";
          shell.dataset.right = maxScroll - scrollArea.scrollLeft > 6 ? "true" : "false";
        }

        function compareValues(a, b, type) {
          if (type === "number") {
            return Number(a) - Number(b);
          }
          return String(a).localeCompare(String(b), undefined, { sensitivity: "base" });
        }

        function applySortIndicators() {
          headers.forEach((header) => {
            const icon = header.querySelector(".sort-icon");
            if (header.dataset.key === state.key) {
              header.setAttribute("aria-sort", state.direction === "asc" ? "ascending" : "descending");
              icon.textContent = state.direction === "asc" ? "↑" : "↓";
            } else {
              header.removeAttribute("aria-sort");
              icon.textContent = "↕";
            }
          });
        }

        function toCamel(s) {
          return s.replace(/-([a-z])/g, function (_, c) { return c.toUpperCase(); });
        }

        function sortRows(key, direction) {
          const header = headers.find((item) => item.dataset.key === key);
          const type = header ? header.dataset.type : "text";
          const prop = toCamel(key);
          const sorted = rows.slice().sort((left, right) => {
            const result = compareValues(left.dataset[prop], right.dataset[prop], type);
            if (result !== 0) {
              return direction === "asc" ? result : -result;
            }
            return left.dataset.project.localeCompare(right.dataset.project);
          });

          sorted.forEach((row) => tbody.appendChild(row));
          state.key = key;
          state.direction = direction;
          applySortIndicators();
        }

        headers.forEach((header) => {
          const button = header.querySelector("button");
          button.addEventListener("click", () => {
            const key = header.dataset.key;
            const direction = state.key === key && state.direction === "asc" ? "desc" : "asc";
            sortRows(key, direction);
          });
        });

        scrollArea.addEventListener("scroll", updateScrollHints, { passive: true });
        window.addEventListener("resize", updateScrollHints);

        applySortIndicators();
        updateScrollHints();
      })();
    </script>
